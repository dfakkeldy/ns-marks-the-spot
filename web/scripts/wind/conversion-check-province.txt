Method
------
* Coordinates were parsed programmatically from pdftotext -layout output of the published PDFs
  (work/extract.py), except Sugar Maple, whose table is an embedded image: it was extracted with
  pdfimages, upscaled and transcribed by eye, and every value was compared with the second copy of
  the same table in Appendix K (identical; Appendix K labels the relocated turbine S10, Appendix D S10R).
* Sugar Maple OCR cross-check of the hand transcription: the two embedded table images were extracted
  (pdfimages), upscaled 4x, binarised and the Easting/Northing columns OCR'd separately with tesseract
  (digits-only whitelist). Appendix K image: 16/16 northings and 14/16 eastings match the typed values;
  the 2 misses (S2, S4) are OCR reading "568" for "558". Appendix D image: 14/16 eastings match; the 2
  misses are OCR errors (S3 "58041", S7 "659745"); D northings did not OCR. Every typed easting is matched
  exactly by at least one OCR pass and every northing by the K pass, so no transcription error remains.
* Strum-format tables (Melvin Lake, Weavers Mountain, Mersey River, Ellershouse 3) appear 8-16 times in
  the same PDF (one per EMI consultation letter); the parser compared every copy: all copies identical.
* Source CRS -> EPSG code used:
    'UTM NAD83 Zone 20' / 'UTM Z20' / 'UTM 20N' (Strum, Clydesdale)  -> EPSG:26920 NAD83 / UTM 20N
    'UTM NAD83 Zone 19' (Wedgeport)                                   -> EPSG:26919 NAD83 / UTM 19N
    'UTM, NAD83(CSRS), Zone 20' (Nortek: Eigg Mountain, Sugar Maple)   -> EPSG:2961 NAD83(CSRS) / UTM 20N
    Benjamins Mill: printed as 'Geo [deg]-WGS84' decimal degrees (6 dp) -> used as printed, no conversion
    Harbour Hills (separate file): printed as degrees-minutes-seconds, datum not stated in the table
      (report maps are NAD83) -> converted DMS->decimal degrees arithmetically, no datum shift
* Target: EPSG:4326 (WGS84), 6 decimals. In PROJ, NAD83 and NAD83(CSRS) -> WGS84 are treated as
  null/ballpark transformations; the real differences here are about 1-2 m, far below the accuracy of a
  pre-construction layout (micrositing commonly moves turbines tens of metres). CHECK 4 below shows
  that reading the same UTM numbers as NAD83 or NAD83(CSRS) changes the result by ~1 m.
* Tools: /opt/local/bin/python3.12 -I with pyproj 3.6.1 (PROJ 9.3.1); independent cs2cs (PROJ 9.9.0).

"""Extract published turbine coordinate tables from pdftotext -layout output.
Writes work/raw_turbines.json: list of dicts with project, turbine, coords_as_published, crs, x, y, page."""
import re, json, sys, os
D = sys.argv[1]  # dl/docs dir
OUT = sys.argv[2]
def pages(fn):
    return open(os.path.join(D, fn), encoding='utf-8', errors='replace').read().split('\f')
recs = []
def add(project, tid, pub, crs, x, y, page, extra=None):
    r = dict(project=project, turbine=tid, coords_as_published=pub, crs=crs, x=x, y=y, page=page)
    if extra: r.update(extra)
    recs.append(r)

# 1. Eigg Mountain: Appendix K Table 1 (page 4)
p = pages('eigg-mountain-wind-project/eigg-mountain-appendix-k.txt')[3]
blk = p.split('Table 1: Proposed Turbine Locations.')[1].split('Table 2:')[0]
for m in re.finditer(r'^\s*(\d{1,2})\s+N163/6\.X\s+([\d,]+)\s+([\d,]+)\s*$', blk, re.M):
    e, n = m.group(2), m.group(3)
    add('eigg-mountain', m.group(1), f'E {e} N {n} (UTM, NAD83(CSRS), Zone 20)', 'EPSG:2961',
        float(e.replace(',', '')), float(n.replace(',', '')), 4)
# 2. Strum EMI-letter tables (Table 1: Proposed Turbine Locations & Specifications)
def strum(project, fn, zone_label):
    ps = pages(fn)
    first = None; sets = []
    for i, p in enumerate(ps):
        if 'Table 1: Proposed Turbine Locations & Specifications' not in p: continue
        blk = p.split('Table 1: Proposed Turbine Locations & Specifications')[1]
        rows = re.findall(r'^\s*(\d{1,2})\s+([\d.]+)\s+([\d.]+)\s+(4[45]\.\d+)\s+(-6\d\.\d+)\s', blk, re.M)
        if not rows: continue
        sets.append((i + 1, tuple(rows)))
    base_page, base = sets[0]
    same = [pg for pg, r in sets if r == base]
    diff = [pg for pg, r in sets if r != base]
    for tid, e, n, lat, lon in base:
        add(project, tid, f'E {e} N {n} ({zone_label}); table also lists lat {lat}, lon {lon}',
            'EPSG:26920', float(e), float(n), base_page, dict(pub_lat=float(lat), pub_lon=float(lon)))
    return same, diff
info = {}
info['melvin-lake'] = strum('melvin-lake', 'melvin-lake-wind-project/mlwp-Part-5-Appendix-G-part-b-to-Appendix-N.txt', 'UTM 20N')

ps = pages('wedgeport-wind-farm-project/wwf-ea-registration-document.txt')
for pgno in (49, 50):

        add('wedgeport', 'WTG' + m.group(1), f'E {m.group(2)} N {m.group(3)} (UTM NAD83 Zone 19)', 'EPSG:26919',

"""Convert published UTM coordinates to WGS84 lon/lat with pyproj and run checks.
Run with: /opt/local/bin/python3.12 -I convert.py raw_turbines.json nrcan_ns.json converted.json"""
import json, sys, math, statistics
import pyproj
from pyproj import Transformer
raw = json.load(open(sys.argv[1]))
nrcan = json.load(open(sys.argv[2]))
out = sys.argv[3]
print('pyproj', pyproj.__version__, 'PROJ', pyproj.proj_version_str)
T = {c: Transformer.from_crs(c, 'EPSG:4326', always_xy=True) for c in ('EPSG:26920', 'EPSG:26919', 'EPSG:2961')}
INV = {c: Transformer.from_crs('EPSG:4326', c, always_xy=True) for c in T}
for c, t in T.items():
    print(c, pyproj.CRS(c).name, '->', 'EPSG:4326 |', t.description)
def hav(lat1, lon1, lat2, lon2):
    R = 6371008.8
    p1, p2 = math.radians(lat1), math.radians(lat2)
    dp, dl = p2 - p1, math.radians(lon2 - lon1)
    a = math.sin(dp / 2) ** 2 + math.cos(p1) * math.cos(p2) * math.sin(dl / 2) ** 2
    return 2 * R * math.asin(math.sqrt(a))
recs = raw['records']
maxrt = 0
for r in recs:
    if r['crs'] == 'EPSG:4326':
        r['lon'], r['lat'] = r['x'], r['y']
        continue
    lon, lat = T[r['crs']].transform(r['x'], r['y'])
    x2, y2 = INV[r['crs']].transform(lon, lat)
    maxrt = max(maxrt, abs(x2 - r['x']), abs(y2 - r['y']))
    r['lon'], r['lat'] = round(lon, 6), round(lat, 6)
print('\nCHECK 1 round trip UTM->WGS84->UTM max residual (m): %.6f' % maxrt)
print('\nCHECK 2 conversion vs lat/lon printed in the same source table')
by = {}
for r in recs: by.setdefault(r['project'], []).append(r)
for p, rs in by.items():
    if 'pub_lat' not in rs[0]: continue
    d = [(hav(r['lat'], r['lon'], r['pub_lat'], r['pub_lon']), r['turbine']) for r in rs]
    dec = min(len(str(r['pub_lat']).split('.')[1]) for r in rs)
    print('  %-17s n=%2d median %.1f m, max %.1f m (turbine %s); printed lat/lon precision ~%d dp' % (
        p, len(d), statistics.median(x for x, _ in d), max(d)[0], max(d)[1], dec))
    for x, t in d:
        if x > 300: print('     outlier: turbine', t, 'differs by %.0f m from its printed lat/lon' % x)
print('\nCHECK 3 independent known points: existing turbines tabulated in the Eigg Mountain shadow-flicker report')
print('  (UTM NAD83(CSRS) Zone 20, Appendix K Tables 2-3) vs NRCan installed-turbine positions (Glen Dhu, Maryvale)')
known = [('Glen Dhu 1', 562130, 5059751), ('Glen Dhu 2', 561800, 5059596), ('Glen Dhu 3', 561769, 5059192),
         ('Glen Dhu 4', 561362, 5059386), ('Glen Dhu 5', 562329, 5059198), ('Glen Dhu 6', 561575, 5058622),
         ('Glen Dhu 7', 561867, 5058457), ('Glen Dhu 8', 562394, 5058428), ('Glen Dhu 9', 562600, 5058049),
         ('Maryvale 1', 572478, 5064309), ('Maryvale 2', 572749, 5064258)]
nr = [(f['attributes']['Project_Name'], f['attributes']['Turbine_Identifier'], f['attributes']['Latitude'], f['attributes']['Longitude'])
      for f in nrcan['features'] if f['attributes']['Project_Name'] in ('Glen Dhu', 'Maryvale')]
for name, e, n in known:
    lon, lat = T['EPSG:2961'].transform(e, n)
    best = min(nr, key=lambda z: hav(lat, lon, z[2], z[3]))
    print('  %-11s -> %.6f, %.6f  nearest NRCan %s %s at %.1f m' % (name, lat, lon, best[0], best[1], hav(lat, lon, best[2], best[3])))
print('\nCHECK 4 datum sensitivity: same UTM numbers read as NAD83(original) vs NAD83(CSRS)')
for r in recs[:1] + [x for x in recs if x['project'] == 'wedgeport'][:1]:
    pass
t_a = Transformer.from_crs('EPSG:26920', 'EPSG:4326', always_xy=True)
t_b = Transformer.from_crs('EPSG:2961', 'EPSG:4326', always_xy=True)
for e, n in ((565656.46, 5043451.31), (430207.92, 4969534.99), (354581, 4881125)):
    a = t_a.transform(e, n); b = t_b.transform(e, n)
    print('  E%.0f N%.0f: 26920 vs 2961 differ by %.2f m' % (e, n, hav(a[1], a[0], b[1], b[0])))
json.dump(recs, open(out, 'w'), indent=1)
